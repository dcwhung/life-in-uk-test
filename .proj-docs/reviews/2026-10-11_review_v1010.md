```js
// js/components/tags.js — "• " remark: the marker sits in a fixed-width box like .note-mark,
// so a wrapped line lines up with the text in any font
const NOTE_CELL_BULLET = /^•\s+/;
function noteCellSubHtml(remark) {
  const mark = remark.match(NOTE_CELL_BULLET);
  if (!mark) return `<span class="note-cell-sub">${escapeHtml(remark)}</span>`;
  return `<span class="note-cell-sub bullet"><span class="note-cell-mark">• </span>${escapeHtml(remark.slice(mark[0].length))}</span>`;
}
```
