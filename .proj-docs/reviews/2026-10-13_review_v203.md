```js
// js/screens/study.js — desktop: a card top inside the panel's top fade also counts as hidden (S-252 option A)
function studyCardTopHidden(el) {
  const top = el.getBoundingClientRect().top;
  if (studyPanelScrolls()) {
    const fade = parseFloat(getComputedStyle(el).scrollMarginTop) || 0; // = --fit-fade-h (css/screens/study.css)
    return top < studyPanel().getBoundingClientRect().top + fade - 1;
  }
  return top < (parseFloat(getComputedStyle(document.documentElement).scrollPaddingTop) || 0) - 1;
}
```
