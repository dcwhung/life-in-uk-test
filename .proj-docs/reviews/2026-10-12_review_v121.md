```js
// ── v1.2.1: merged STUDY facts (FACT_ID_MERGES) — move fact-id keyed progress to the kept id ──
function migrateFactIds() {
  if (typeof FACT_ID_MERGES === 'undefined') return;
  const merges = Object.entries(FACT_ID_MERGES).map(([oldId, keptId]) => [Number(oldId), keptId]);
  // W-057: the saved plan is checked on every load (old ids are never reused, so this is idempotent): a tab still
  // running an older version can save its in-memory plan, old ids included, after the marker was written
  rewriteStoredJson(STUDY_PLAN_LS, v => mergePlanFacts(v, merges));
  const done = mergedFactIds();
  const todo = merges.filter(([oldId]) => !done.includes(oldId));
  if (!todo.length) return;
  const moved = FACT_ID_MAP_LS.flatMap(key => storedNames(key).map(name => rewriteStoredJson(name, v => mergeFactMap(v, todo))));
  if (moved.some(Boolean)) localStorage.setItem(FACT_MERGE_LS, JSON.stringify([...done, ...todo.map(([oldId]) => oldId)]));
}
// S-195: while a legacy fallback is in use the stale new key may still be merged back in later: rewrite both names
function storedNames(key) { return [...new Set([key, LS_KEY_FALLBACK[key] || key])]; }
// rewriteStoredJson then takes the storage name as it is (no LS_KEY_FALLBACK lookup of its own)

// the saved plan's tasks list fact ids: old → kept; W-056: a read task drops the old id when another read task
// already lists the kept id (unless that would empty it); S-196: a wrong-facts anchor moves with its fact
function mergePlanFacts(plan, todo) {
  if (!plan || !Array.isArray(plan.days)) return undefined;
  const ids = new Map(todo);
  const tasks = plan.days.flatMap(day => (Array.isArray(day && day.tasks) ? day.tasks : []))
    .filter(task => task && Array.isArray(task.facts));
  const readIds = new Set(tasks.filter(t => t.type === PLAN_TASK.read).flatMap(t => t.facts));
  let changed = false;
  tasks.filter(task => task.facts.some(id => ids.has(id))).forEach(task => {
    const elsewhere = id => task.type === PLAN_TASK.read && readIds.has(ids.get(id)) && !task.facts.includes(ids.get(id));
    const kept = task.facts.filter(id => !ids.has(id) || !elsewhere(id)).map(id => (ids.has(id) ? ids.get(id) : id));
    task.facts = kept.length ? [...new Set(kept)] : [...new Set(task.facts.map(id => (ids.has(id) ? ids.get(id) : id)))];
    moveAnchor(task, ids);
    changed = true;
  });
  return changed ? plan : undefined;
}
function moveAnchor(task, ids) {
  if (!task.anchor || typeof task.anchor !== 'object') return;
  ids.forEach((keptId, oldId) => {
    if (oldId in task.anchor && !(keptId in task.anchor)) task.anchor[keptId] = task.anchor[oldId];
    delete task.anchor[oldId];
  });
}
```

> 注意：`PLAN_TASK` 喺 `js/domain/plan.js` 定義，載入次序喺 `utils.js` 之後，但 `migrateFactIds()` 係第一次 `getLS()` 先行（lazy），所以要確認嗰陣 `PLAN_TASK` 已經存在；如果唔肯定，就喺 `config.js` 寫 `'read'` 常數，或者用 `typeof PLAN_TASK` guard。
